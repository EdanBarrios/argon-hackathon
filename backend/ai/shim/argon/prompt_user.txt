```c
/* If type == XML_CTYPE_EMPTY or XML_CTYPE_ANY, then quant will be
   XML_CQUANT_NONE, and the other fields will be zero or NULL.
   If type == XML_CTYPE_MIXED, then quant will be NONE or REP and
   numchildren will contain number of elements that may be mixed in
   and children point to an array of XML_Content cells that will be
   all of XML_CTYPE_NAME type with no quantification.

   If type == XML_CTYPE_NAME, then the name points to the name, and
   the numchildren field will be zero and children will be NULL. The
   quant fields indicates any quantifiers placed on the name.

   CHOICE and SEQ will have name NULL, the number of children in
   numchildren and children will point, recursively, to an array
   of XML_Content cells.

   The EMPTY, ANY, and MIXED types will only occur at top level.
*/

  enum XML_Content_Type type;
  enum XML_Content_Quant quant;

  unsigned int numchildren;
  XML_Content *children;
};

XMLPARSEAPI(XML_Parser)

/* Constructs a new parser and namespace processor.  Element type
   names and attribute names that belong to a namespace will be
   expanded; unprefixed attribute names are never expanded; unprefixed
   element type names are expanded only if there is a default
   namespace. The expanded name is the concatenation of the namespace
   URI, the namespace separator character, and the local part of the
   name.  If the namespace separator is '\0' then the namespace URI
   and the local part will be concatenated without any separator.
   It is a programming error to use the separator '\0' with namespace
   triplets (see XML_SetReturnNSTriplet).
   If a namespace separator is chosen that can be part of a URI or
   part of an XML name, splitting an expanded name back into its
   1, 2 or 3 original parts on application level in the element handler
   may end up vulnerable, so these are advised against;  sane choices for
   a namespace separator are e.g. '\n' (line feed) and '|' (pipe).

   Note that Expat does not validate namespace URIs (beyond encoding)
   against RFC 3986 today (and is not required to do so with regard to
   the XML 1.0 namespaces specification) but it may start doing that
   in future releases.  Before that, an application using Expat must
   be ready to receive namespace URIs containing non-URI characters.
*/
XMLPARSEAPI(XML_Parser)
XML_ParserCreateNS(const XML_Char *encoding, XML_Char namespaceSeparator);

/* Constructs a new parser using the memory management suite referred to
   by memsuite. If memsuite is NULL, then use the standard library memory
   suite. If namespaceSeparator is non-NULL it creates a parser with
   namespace processing as described above. The character pointed at
   will serve as the namespace separator.

   All further memory operations used for the created parser will come from
   the given suite.
*/
XMLPARSEAPI(XML_Parser)
XML_ParserCreate_MM(const XML_Char *encoding,
                    const XML_Memory_Handling_Suite *memsuite,
                    const XML_Char *namespaceSeparator);

   Added in Expat 1.95.3.

   XML_ExternalEntityParserCreate.

/* This structure is filled in by the XML_UnknownEncodingHandler to
   provide information to the parser about encodings that are unknown
   to the parser.

   The map[b] member gives information about byte sequences whose
   first byte is b.

   If map[b] is c where c is >= 0, then b by itself encodes the
   Unicode scalar value c.

   If map[b] is -1, then the byte sequence is malformed.

   If map[b] is -n, where n >= 2, then b is the first byte of an
   n-byte sequence that encodes a single Unicode scalar value.

   The data member will be passed as the first argument to the convert
   function.

   The convert function is used to convert multibyte sequences; s will
   point to a n-byte sequence where map[(unsigned char)*s] == -n.  The
   convert function must return the Unicode scalar value represented
   by this byte sequence or -1 if the byte sequence is malformed.

   The convert function may be NULL if the encoding is a single-byte
   encoding, that is if map[b] >= -1 for all bytes b.

   When the parser is finished with the encoding, then if release is
   not NULL, it will call release passing it the data member; once
   release has been called, the convert function will not be called
   again.

   Expat places certain restrictions on the encodings that are supported
   using this mechanism.

   1. Every ASCII character that can appear in a well-formed XML document,
      other than the characters

      $@\^`{}~

      must be represented by a single byte, and that byte must be the
      same byte that represents that character in ASCII.

   2. No character may require more than 4 bytes to encode.

   3. All characters encoded must have Unicode scalar values <=
      0xFFFF, (i.e., characters that would be encoded by surrogates in
      UTF-16 are  not allowed).  Note that this restriction doesn't
      apply to the built-in support for UTF-8 and UTF-16.

   4. No Unicode character may be encoded by more than one distinct
      sequence of bytes.
*/
typedef struct {
  int map[256];
  void *data;
  int(XMLCALL *convert)(void *data, const char *s);
  void(XMLCALL *release)(void *data);
} XML_Encoding;

/* This is called for an encoding that is unknown to the parser.

   The encodingHandlerData argument is that which was passed as the
   second argument to XML_SetUnknownEncodingHandler.

   The name argument gives the name of the encoding as specified in

   If the callback can provide information about the encoding, it must
   fill in the XML_Encoding structure, and return XML_STATUS_OK.
   Otherwise it must return XML_STATUS_ERROR.

   If info does not describe a suitable encoding, then the parser will
   return an XML_ERROR_UNKNOWN_ENCODING error.
*/
typedef int(XMLCALL *XML_UnknownEncodingHandler)(void *encodingHandlerData,
                                                 const XML_Char *name,
                                                 XML_Encoding *info);

/* If do_nst is non-zero, and namespace processing is in effect, and
   a name has a prefix (i.e. an explicit namespace qualifier) then
   that name is returned as a triplet in a single string separated by
```

## The C driver used to verify expat.rs: which C callback each Event comes from
```c
/*
 * Differential-test driver for libexpat (Theseus harness, role C).
 *
 * Reads all of stdin, creates a parser with XML_ParserCreate(NULL) (encoding
 * from the BOM or XML declaration, UTF-8 otherwise), sets a fixed all-zero
 * hash salt, and calls XML_Parse once with isFinal = 1. Every callback
 * appends one event line to an in-memory buffer. Then:
 *   success: print all event lines, then the line OK
 *   failure: print ONLY the line ERROR <code> line <L> col <C>
 *            (events seen before the error are discarded)
 * where <code> is XML_GetErrorCode (an enum XML_Error value) and L / C come
 * from XML_GetCurrentLineNumber64 / XML_GetCurrentColumnNumber64.
 * Always exits 0 unless the process crashes or a sanitizer aborts it.
 *
 * Scope: no namespace processing (XML_ParserCreate, not ...NS), no external
 * entity loading (no external entity ref handler, parameter entity parsing
 * left at the default XML_PARAM_ENTITY_PARSING_NEVER), internal DTD entities
 * expanded, billion-laughs protection at library defaults.
 *
 * Event lines (one per callback, in callback order):
 *   XMLDECL <version> <encoding> <standalone>
 *   DOCTYPE_START <name> <sysid> <pubid> <has_internal_subset>
 *   DOCTYPE_END
 *   ENTITY <name> <is_pe> VALUE <value>
 *   ENTITY <name> <is_pe> EXTERNAL <sysid> <pubid> <notation>
 *   START <name>[ <attr>=<value>]...   attributes in the order expat gives
 *                                       them (specified first, then defaulted)
 *   END <name>
 *   TEXT <text>                          adjacent character data coalesced
 *   PI <target> <data>
 *   COMMENT <text>
 *   CDATA_START
 *   CDATA_END
 *   SKIPPED <name> <is_pe>
 * Every string field is printed as a quoted string ("...") or as the bare
 * word null when expat passes NULL. Inside quotes, bytes 0x20..0x7E print as
 * themselves except '"' and '\\', which print as \" and \\; every other byte
 * prints as \xNN (two lowercase hex digits). Integers print in decimal.
 * Pending TEXT is flushed before any other event and before the final line.
 */
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#include "expat.h"

/* ---- output buffer ----------------------------------------------------- */
/* All event lines go to this buffer. It is written to stdout only if the
   parse succeeds; on error only the ERROR line is printed. This matches the
   Rust interface parse(&[u8]) -> Result<Vec<Event>, XmlError>. */

static char *out_buf = NULL;
static size_t out_len = 0;
static size_t out_cap = 0;

static void out_bytes(const char *s, size_t n)
{
    if (out_len + n > out_cap) {
        size_t cap = out_cap ? out_cap : 65536;
        char *grown;
        while (cap < out_len + n) {
            cap *= 2;
        }
        grown = (char *)realloc(out_buf, cap);
        if (grown == NULL) {
            fputs("driver: out of memory\n", stderr);
            exit(2);
        }
        out_buf = grown;
        out_cap = cap;
    }
    memcpy(out_buf + out_len, s, n);
    out_len += n;
}

static void out_char(char c)
{
    out_bytes(&c, 1);
}

static void out_cstr(const char *s)
{
    out_bytes(s, strlen(s));
}

static void out_int(int v)
{
    char tmp[32];
    int n = snprintf(tmp, sizeof tmp, "%d", v);
    out_bytes(tmp, (size_t)n);
}

/* ---- output helpers ---------------------------------------------------- */

static void put_quoted_n(const char *s, size_t n)
{
    static const char hex[] = "0123456789abcdef";
    size_t i;
    out_char('"');
    for (i = 0; i < n; i++) {
        unsigned char c = (unsigned char)s[i];
        if (c == '"' || c == '\\') {
            out_char('\\');
            out_char(c);
        } else if (c >= 0x20 && c <= 0x7e) {
            out_char(c);
        } else {
            out_char('\\');
            out_char('x');
            out_char(hex[c >> 4]);
            out_char(hex[c & 15]);
        }
    }
    out_char('"');
}

static void put_str(const XML_Char *s)
{
    if (s == NULL) {
        out_cstr("null");
    } else {
        put_quoted_n(s, strlen(s));
    }
}

/* ---- character data coalescing ----------------------------------------- */

static char *text_buf = NULL;
static size_t text_len = 0;
static size_t text_cap = 0;
static int text_pending = 0;

static void flush_text(void)
{
    if (!text_pending) {
        return;
    }
    out_cstr("TEXT ");
    put_quoted_n(text_buf, text_len);
    out_char('\n');
    text_len = 0;
    text_pending = 0;
}

static void XMLCALL on_text(void *ud, const XML_Char *s, int len)
{
    (void)ud;
    if (len <= 0) { /* empty runs never produce a TEXT line */
        return;
    }
    if (text_len + (size_t)len > text_cap) {
        size_t cap = text_cap ? text_cap : 4096;
        char *grown;
        while (cap < text_len + (size_t)len) {
            cap *= 2;
        }
        grown = (char *)realloc(text_buf, cap);
        if (grown == NULL) {
            fputs("driver: out of memory\n", stderr);
            exit(2);
        }
        text_buf = grown;
        text_cap = cap;
    }
    memcpy(text_buf + text_len, s, (size_t)len);
    text_len += (size_t)len;
    text_pending = 1;
}

/* ---- other callbacks ---------------------------------------------------- */

static void XMLCALL on_xmldecl(void *ud, const XML_Char *version,
                               const XML_Char *encoding, int standalone)
{
    (void)ud;
    flush_text();
    out_cstr("XMLDECL ");
    put_str(version);
    out_char(' ');
    put_str(encoding);
    out_char(' '); out_int(standalone); out_char('\n');
}

static void XMLCALL on_doctype_start(void *ud, const XML_Char *name,
                                     const XML_Char *sysid, const XML_Char *pubid,
                                     int has_internal_subset)
{
    (void)ud;
    flush_text();
    out_cstr("DOCTYPE_START ");
    put_str(name);
    out_char(' ');
    put_str(sysid);
    out_char(' ');
    put_str(pubid);
    out_char(' '); out_int(has_internal_subset); out_char('\n');
}

static void XMLCALL on_doctype_end(void *ud)
{
    (void)ud;
    flush_text();
    out_cstr("DOCTYPE_END\n");
}

static void XMLCALL on_entity_decl(void *ud, const XML_Char *name, int is_pe,
                                   const XML_Char *value, int value_length,
                                   const XML_Char *base, const XML_Char *sysid,
                                   const XML_Char *pubid, const XML_Char *notation)
{
    (void)ud;
    (void)base; /* never set: the driver does not call XML_SetBase */
    flush_text();
    out_cstr("ENTITY ");
    put_str(name);
    out_char(' '); out_int(is_pe); out_char(' ');
    if (value != NULL) {
        out_cstr("VALUE ");
        put_quoted_n(value, value_length > 0 ? (size_t)value_length : 0);
    } else {
        out_cstr("EXTERNAL ");
        put_str(sysid);
        out_char(' ');
        put_str(pubid);
        out_char(' ');
        put_str(notation);
    }
    out_char('\n');
}

static void XMLCALL on_start(void *ud, const XML_Char *name, const XML_Char **atts)
{
    int i;
    (void)ud;
    flush_text();
    out_cstr("START ");
    put_str(name);
    for (i = 0; atts[i] != NULL; i += 2) {
        out_char(' ');
        put_str(atts[i]);
        out_char('=');
        put_str(atts[i + 1]);
    }
    out_char('\n');
}

static void XMLCALL on_end(void *ud, const XML_Char *name)
{
    (void)ud;
    flush_text();
    out_cstr("END ");
    put_str(name);
    out_char('\n');
}

static void XMLCALL on_pi(void *ud, const XML_Char *target, const XML_Char *data)
{
    (void)ud;
    flush_text();
    out_cstr("PI ");
    put_str(target);
    out_char(' ');
    put_str(data);
    out_char('\n');
}

static void XMLCALL on_comment(void *ud, const XML_Char *data)
{
    (void)ud;
    flush_text();
    out_cstr("COMMENT ");
    put_str(data);
    out_char('\n');
}

static void XMLCALL on_cdata_start(void *ud)
{
    (void)ud;
    flush_text();
    out_cstr("CDATA_START\n");
}

static void XMLCALL on_cdata_end(void *ud)
{
    (void)ud;
    flush_text();
    out_cstr("CDATA_END\n");
}

static void XMLCALL on_skipped(void *ud, const XML_Char *name, int is_pe)
{
    (void)ud;
    flush_text();
    out_cstr("SKIPPED ");
    put_str(name);
    out_char(' '); out_int(is_pe); out_char('\n');
}

/* ---- main --------------------------------------------------------------- */

static char *read_stdin(size_t *out_len)
{
    size_t cap = 65536;
    size_t len = 0;
    char *buf = (char *)malloc(cap);
    if (buf == NULL) {
        return NULL;
    }
    for (;;) {
        size_t n;
        if (len == cap) {
            char *grown;
            cap *= 2;
            grown = (char *)realloc(buf, cap);
            if (grown == NULL) {
                free(buf);
                return NULL;
            }
            buf = grown;
        }
        n = fread(buf + len, 1, cap - len, stdin);
        len += n;
        if (n == 0) {
            break;
        }
    }
    if (ferror(stdin)) {
        free(buf);
        return NULL;
    }
    *out_len = len;
    return buf;
}

int main(void)
{
    static const uint8_t zero_salt[16] = {0};
    size_t len = 0;
    char *input = read_stdin(&len);
    XML_Parser parser;
    enum XML_Status status;

    if (input == NULL || len > (size_t)0x7fffffff) {
        fputs("driver: failed to read stdin\n", stderr);
        return 2;
    }

    parser = XML_ParserCreate(NULL);
    if (parser == NULL) {
        fputs("driver: XML_ParserCreate failed\n", stderr);
        return 2;
    }
    /* Fixed salt: no entropy read, identical hashing on every run. */
    XML_SetHashSalt16Bytes(parser, zero_salt);

    XML_SetXmlDeclHandler(parser, on_xmldecl);
    XML_SetDoctypeDeclHandler(parser, on_doctype_start, on_doctype_end);
    XML_SetEntityDeclHandler(parser, on_entity_decl);
    XML_SetElementHandler(parser, on_start, on_end);
    XML_SetCharacterDataHandler(parser, on_text);
    XML_SetProcessingInstructionHandler(parser, on_pi);
    XML_SetCommentHandler(parser, on_comment);
    XML_SetCdataSectionHandler(parser, on_cdata_start, on_cdata_end);
    XML_SetSkippedEntityHandler(parser, on_skipped);

    status = XML_Parse(parser, input, (int)len, 1);
    flush_text();
    if (status == XML_STATUS_OK) {
        out_cstr("OK\n");
        fwrite(out_buf, 1, out_len, stdout);
    } else {
        /* Events seen before the error are discarded on purpose. */
        printf("ERROR %d line %llu col %llu\n", (int)XML_GetErrorCode(parser),
               (unsigned long long)XML_GetCurrentLineNumber64(parser),
               (unsigned long long)XML_GetCurrentColumnNumber64(parser));
    }
    fflush(stdout);

    XML_ParserFree(parser);
    free(text_buf);
    free(out_buf);
    free(input);
    return 0;
}
```

Notes that matter for pyexpat behaviour:

- **Namespace parsers come through `XML_ParserCreate_MM`, not `XML_ParserCreateNS`.**
  Verified: `XML_ParserCreateNS` is not referenced by pyexpat.c or _elementtree.c in
  v3.13.5. `pyexpat.ParserCreate(namespace_separator=...)` and minidom/expatbuilder pass
  the separator as `XML_ParserCreate_MM`'s third arg; `_elementtree.c:3673` calls
  `ParserCreate_MM(encoding, &ExpatMemoryHandler, "}")` through the capsule, so
  **every ElementTree parser is a namespace parser**. Agreed behavior (option c): when
  `nsSep != NULL`, the parser returns a clean `XML_STATUS_ERROR` as soon as the document
  has an `xmlns`/`xmlns:*` attribute or a prefixed element/attribute name; namespace-free
  documents parse normally and names are passed through unchanged. Exporting
  `XML_ParserCreateNS` as well is harmless (other consumers may want it) but not required.
- The memsuite argument may be ignored.
- **Replay discipline**: re-read the handler pointer from the parser struct before every
  event (pyexpat clears/swaps handlers mid-parse), and check a stopped flag after every
  callback. pyexpat calls `XML_StopParser` when a Python handler raises; the shim must
  then stop delivering events and return `XML_STATUS_ERROR` with
  `XML_GetErrorCode == XML_ERROR_ABORTED`.
- Handlers receive the `userData` set by `XML_SetUserData`. Attribute arrays passed to
  start-element are NULL-terminated `name, value, name, value, ..., NULL`.
- `XML_ExpatVersion()` must return a string that identifies the shim (e.g.
  `"expat_2.9.0-theseus-argon"`), and `XML_ExpatVersionInfo()` should return `{2,9,0}`,
  so `pyexpat.EXPAT_VERSION` proves which library is loaded.
- `XML_ErrorString(code)` must return a non-NULL static string for every code pyexpat
  can see (it builds `pyexpat.errors` from codes 1..N at import time; NULL is fine
  only past the last code).
- `XML_GetFeatureList()` must return a list terminated by `XML_FEATURE_END`.
- `XML_GetErrorCode`, `XML_GetCurrentLineNumber/ColumnNumber/ByteIndex` must be
  correct after a failed `XML_Parse`; pyexpat builds `ExpatError` from them.
- `XML_GetBuffer(p, len)` + `XML_ParseBuffer(p, len, isFinal)` are used by
  `ParseFile`; the shim must hand back a writable buffer of at least `len` bytes.
- `XML_SetHashSalt` is called unconditionally; a no-op is acceptable.
- `XML_FreeContentModel` is only reached if an element-decl handler fires.

## Rules

1. **Argon's core `expat.rs` stays byte-identical** to the Argon-produced artifact it
   was copied from, checked by sha256 in `build.sh`. Source: the final passing round of
   `backend/ai/results/expat-long-decode/argon-fresh-b/` (the one that replaces an
   `unwrap` with an error return). If that file is not on main yet, fall back to
   `argon-fresh-a/round_3/expat.rs` and say so. No edits, not even formatting.
2. **`unsafe` lives only in the FFI layer** (the shim crate's `ffi.rs` or equivalent):
   raw pointer conversion, `CStr`, callbacks into C, the `#[no_mangle]` exports. The
   core parser stays `#![forbid(unsafe_code)]`-clean.
3. pyexpat.c and _elementtree.c are compiled from CPython v3.13.5 unmodified.
4. Results are reported as measured: tests run/passed/failed/errored/skipped per suite,
   ours vs the stock pyexpat baseline. No inflation.

## Symbols checked against 2.9.0's expat.h

Present in 2.9.0 but **not** referenced by v3.13.5 pyexpat.c, so not required:
`XML_ParserCreateNS`, `XML_SetHashSalt16Bytes`,
`XML_SetBillionLaughsAttackProtectionMaximumAmplification`,
`XML_SetBillionLaughsAttackProtectionActivationThreshold`,
`XML_SetAllocTrackerMaximumAmplification`, `XML_SetAllocTrackerActivationThreshold`.
(Later 3.13.x/3.14 pyexpat references some of these; we pin v3.13.5.)

## Knowingly unsupported (expected test failures)

- Namespaced documents in a namespace parser (option c above: clean error). This
  includes any ElementTree input with `xmlns`.
- Positions during callbacks: `CurrentLineNumber`/`ColumnNumber`/`ByteIndex` are only
  meaningful after an error.
- Incremental output: `iterparse` / `XMLPullParser` yield events only at close.
- `GetSpecifiedAttributeCount` / `specified_attributes` count defaulted attributes too.
- Handlers with no corresponding Argon event never fire.

- `XML_SetReturnNSTriplet` and namespace decl handlers.
- Incremental parsing semantics: the shim may buffer all input until `isFinal` and
  parse once, so handlers fire late and position/`buffer_text` tests may differ.
- `XML_StopParser` / resume and `XML_SetReparseDeferralEnabled` (stored, ignored).
- External entities (`XML_ExternalEntityParserCreate` may return NULL,
  `XML_SetParamEntityParsing` stored, ignored), `XML_UseForeignDTD`.
- DTD content models (`XML_SetElementDeclHandler` may never fire), attlist/entity/
  notation decl handlers where the Rust core does not surface them.
- Unknown encodings (`XML_SetUnknownEncodingHandler` ignored), `XML_GetInputContext`
  may return NULL.

## Build and run

Rust side (ai): `backend/ai/shim/build.sh` -> `runs/shim/libtheseus_expat.dylib`.

Python side (systems, branch `feat/systems-pyexpat`):

```
backend/harness/pyexpat/fetch_cpython.sh                 # v3.13.5 sources + tests
backend/harness/pyexpat/build_pyexpat.sh runs/shim/libtheseus_expat.dylib runs/pyexpat/build
PYTHONPATH=runs/pyexpat/build python3 -c 'import pyexpat; print(pyexpat.__file__, pyexpat.EXPAT_VERSION)'
backend/harness/pyexpat/run_tests.sh runs/pyexpat/build   # ours + baseline summary
```

`build_pyexpat.sh` links the dylib by path with `-Wl,-rpath,<dylib dir>`, so the
`@rpath` install name resolves without `DYLD_LIBRARY_PATH`. It also rebuilds
`_elementtree.c` against the same expat.h, because the stock `_elementtree` rejects
a pyexpat capsule whose `XML_MINOR_VERSION` differs from its own (2.7 vs 2.9).
Symbol list regeneration: `build_pyexpat.sh --object-only <dir>; nm -u <dir>/pyexpat.o | grep XML_`.
